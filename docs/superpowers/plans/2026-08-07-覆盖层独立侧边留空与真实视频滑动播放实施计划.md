# 覆盖层独立侧边留空与真实视频滑动播放实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让每个边缘覆盖层拥有独立的侧边留空配置，并将“显示视频”升级为可循环播放 HTTPS 直链、跟随手指上下拖动并在阈值后自动翻页的真实短视频覆盖层。

**Architecture:** 将 `edgeInsetDp` 纳入既有 `EdgeZoneConfig`，用每层的 SharedPreferences 键存储，并在普通悬浮窗、原生触摸和展开聊天态按同一配置计算命中区域。视频覆盖层保留现有顶栏和操作交互，改为两个原生 `TextureView` 页面与两个按需创建的 `MediaPlayer`，当前页和相邻页随拖动位移，提交后交换页面并释放旧播放器。

**Tech Stack:** Kotlin、Android Accessibility Overlay、WindowManager、Compose、TextureView、MediaPlayer、JUnit 4、Robolectric。

```kotlin
data class EdgeZoneConfig(
    val side: EdgeSide,
    val zoneId: Int = DEFAULT_ZONE_ID,
    val enabled: Boolean = true,
    val thicknessDp: Int = 24,
    val edgeInsetDp: Int = 0,
    val topInsetPercent: Int = 15,
    val bottomInsetPercent: Int = 15
)

fun sanitize(config: EdgeZoneConfig): EdgeZoneConfig {
    val zoneId = config.zoneId.coerceIn(0, MAX_ZONES_PER_SIDE - 1)
    val topInset = config.topInsetPercent.coerceIn(0, MAX_INSET_PERCENT)
    val bottomInset = config.bottomInsetPercent.coerceIn(0, MAX_INSET_PERCENT - topInset)
    return config.copy(
        zoneId = zoneId,
        thicknessDp = config.thicknessDp.coerceIn(MIN_THICKNESS_DP, MAX_THICKNESS_DP),
        edgeInsetDp = sanitizeEdgeInsetDp(config.edgeInsetDp),
        topInsetPercent = topInset,
        bottomInsetPercent = bottomInset
    )
}
```

```kotlin
private fun edgeConfig(side: EdgeSide, zoneId: Int): EdgeZoneConfig {
    val inset = if (prefs.contains("${zonePrefix}_edge_inset_dp")) {
        prefs.getInt("${zonePrefix}_edge_inset_dp", defaultConfig.edgeInsetDp)
    } else if (zoneId < edgeZoneCount(side)) {
        prefs.getInt(edgeInsetKey(side), defaultConfig.edgeInsetDp)
    } else {
        defaultConfig.edgeInsetDp
    }
    return EdgeZoneConfig(
        side = side,
        zoneId = zoneId,
        enabled = enabled,
        thicknessDp = thicknessDp,
        edgeInsetDp = inset,
        topInsetPercent = topInsetPercent,
        bottomInsetPercent = bottomInsetPercent
    ).sanitized()
}

private fun putEdgeConfig(editor: SharedPreferences.Editor, config: EdgeZoneConfig) {
    editor.putInt("${edgeZonePrefix(config.side, config.zoneId)}_edge_inset_dp", config.sanitized().edgeInsetDp)
    // Existing zone fields remain unchanged.
}
```

```kotlin
fun videoDemoPageIndex(currentIndex: Int, direction: VideoDemoPageDirection, itemCount: Int): Int {
    if (itemCount <= 0) return 0
    val safeIndex = Math.floorMod(currentIndex, itemCount)
    return when (direction) {
        VideoDemoPageDirection.Next -> (safeIndex + 1) % itemCount
        VideoDemoPageDirection.Previous -> Math.floorMod(safeIndex - 1, itemCount)
    }
}
```

```kotlin
internal fun textureViewScaleForAspectFill(
    containerWidth: Int,
    containerHeight: Int,
    videoAspectRatio: Float?
): Pair<Float, Float> {
    if (containerWidth <= 0 || containerHeight <= 0 || videoAspectRatio == null || videoAspectRatio <= 0f) return 1f to 1f
    val containerRatio = containerWidth.toFloat() / containerHeight
    return if (videoAspectRatio > containerRatio) videoAspectRatio / containerRatio to 1f else 1f to containerRatio / videoAspectRatio
}
```

Apply this transform when the texture size or media video size changes. Keep the existing AspectFit helper unchanged because it belongs to a separate media preview surface.
